# dyno-extract-sdk@0.1.0-alpha.1

A TypeScript SDK client for the api.dyno.hideyoshi.sh API.

## Usage

First, install the SDK from npm.

```bash
npm install dyno-extract-sdk --save
```

Next, try it out.


```ts
import {
  Configuration,
  BlueprintApi,
} from 'dyno-extract-sdk';
import type { AddRevisionRequest } from 'dyno-extract-sdk';

async function example() {
  console.log("🚀 Testing dyno-extract-sdk SDK...");
  const config = new Configuration({ 
    // To configure API key authorization: APIKeyHeader
    apiKey: "YOUR API KEY",
  });
  const api = new BlueprintApi(config);

  const body = {
    // string
    slug: slug_example,
    // { [key: string]: any | null; }
    requestBody: Object,
  } satisfies AddRevisionRequest;

  try {
    const data = await api.addRevision(body);
    console.log(data);
  } catch (error) {
    console.error(error);
  }
}

// Run the test
example().catch(console.error);
```
